const url = `https://world.openbeautyfacts.org/cgi/search.pl?search_terms=${encodeURIComponent(searchQ)}&search_simple=1&action=process&json=1&page_size=8&fields=product_name,product_name_en,brands,ingredients_text,ingredients_text_en`; const res = await fetch(url, { headers: { 'Accept': 'application/json' } }); if (!res.ok) throw new Error("API error"); const data = await res.json(); if(data.products && data.products.length > 0){ const withIngredients = data.products.filter(p => (p.ingredients_text_en && p.ingredients_text_en.length > 10) || (p.ingredients_text && p.ingredients_text.length > 10) ).slice(0, 4); if(withIngredients.length > 0){ setSearchResults(withIngredients); } else { setSearchResults(data.products.slice(0, 4)); setSearchError("Found products but ingredient lists may be incomplete. Paste ingredients manually if needed."); } } else { setSearchError("No products found. Try a shorter name or paste the ingredient list directly."); }
